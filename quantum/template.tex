%command to define new quantities, argument order is: handle, color, symbol
\newcommand{\newquant}[3]{

    %patch the diff command
    \patchcommand\diff{\ifthenelse{\equal{#1}{##1}}{\color{#2}}{}}{\ifthenelse{\equal{#1}{##1}}{#1}{}}

    %patch the pdiff command
    \patchcommand\ddiff{\ifthenelse{\equal{#1}{##1}}{\color{#2}}{}}{\ifthenelse{\equal{#1}{##1}}{#1}{}}

    %patch the qdiff command
    \patchcommand\pdiff{\ifthenelse{\equal{#1}{##1}}{\color{#2}}{}}{\ifthenelse{\equal{#1}{##1}}{#1}{}}

    %patch the change command
    \patchcommand\change{\ifthenelse{\equal{#1}{##1}}{\color{#2}}{}}{\ifthenelse{\equal{#1}{##1}}{#1}{}}
    
    %add the symbol as a command    
    \expandarg
    \exploregroups
        %to avoid double subscripts
        \IfSubStr{#3}{_}{ 
            \newcommand{#1}[1][]{
                \color{#2}{\StrBefore{#3}{_}_{\StrBehind{#3}{_} ##1}}\color{black}
                }
        }{
            \newcommand{#1}[1][]{
                \color{#2}{#3_{##1}}\color{black}
                }
        }
        
}
